    <script>
        // Navegación SPA simulada
        document.addEventListener('DOMContentLoaded', function() {
            const navLinks = document.querySelectorAll('nav a');
            const sections = document.querySelectorAll('.section');
            
            // Configurar eventos de clic para navegación
            navLinks.forEach(link => {
                link.addEventListener('click', function(e) {
                    e.preventDefault();
                    
                    // Actualizar navegación activa
                    navLinks.forEach(l => l.classList.remove('active'));
                    this.classList.add('active');
                    
                    // Obtener la sección objetivo
                    const targetSection = this.getAttribute('data-section');
                    
                    // Ocultar todas las secciones
                    sections.forEach(section => {
                        section.classList.remove('active');
                    });
                    
                    // Mostrar la sección objetivo
                    document.getElementById(targetSection).classList.add('active');
                });
            });
            
            // Interacción de hipermedia: mostrar/ocultar información adicional
            const toggleButton = document.getElementById('toggle-info');
            if (toggleButton) {
                const hiddenContent = document.getElementById('hidden-info');
                
                toggleButton.addEventListener('click', function() {
                    if (hiddenContent.style.display === 'block') {
                        hiddenContent.style.display = 'none';
                        toggleButton.textContent = 'Ver línea de tiempo interactiva';
                    } else {
                        hiddenContent.style.display = 'block';
                        toggleButton.textContent = 'Ocultar línea de tiempo';
                    }
                });
            }
            
            // =============================================
            // IMPLEMENTACIÓN DE LA EVALUACIÓN Nro. 2
            // =============================================
            
            // 2. FEEDBACK INTERACTIVO - Validación en tiempo real
            const contactForm = document.getElementById('contactForm');
            const nameInput = document.getElementById('name');
            const emailInput = document.getElementById('email');
            const subjectInput = document.getElementById('subject');
            const messageInput = document.getElementById('message');
            const submitBtn = document.getElementById('submitBtn');
            
            // 3. FUNCIÓN DE UTILIDAD - Función modular para validación
            function validateField(value, fieldType, minLength = 2) {
                // Esta es nuestra función modular reutilizable
                if (!value || value.trim().length < minLength) {
                    return {
                        isValid: false,
                        message: `Este campo debe tener al menos ${minLength} caracteres`
                    };
                }
                
                // Validación específica por tipo de campo
                switch(fieldType) {
                    case 'email':
                        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
                        if (!emailRegex.test(value)) {
                            return {
                                isValid: false,
                                message: 'Por favor ingresa un email válido'
                            };
                        }
                        break;
                        
                    case 'name':
                        if (!/^[a-zA-ZáéíóúÁÉÍÓÚñÑ\s]+$/.test(value)) {
                            return {
                                isValid: false,
                                message: 'El nombre solo puede contener letras y espacios'
                            };
                        }
                        break;
                }
                
                return {
                    isValid: true,
                    message: '✓ Campo válido'
                };
            }
            
            // Función para actualizar el estado del formulario
            function updateFormState() {
                const nameValidation = validateField(nameInput.value, 'name');
                const emailValidation = validateField(emailInput.value, 'email');
                const subjectValidation = validateField(subjectInput.value, 'text', 3);
                const messageValidation = validateField(messageInput.value, 'text', 10);
                
                const isFormValid = nameValidation.isValid && 
                                  emailValidation.isValid && 
                                  subjectValidation.isValid && 
                                  messageValidation.isValid;
                
                submitBtn.disabled = !isFormValid;
                return isFormValid;
            }
            
            // Eventos de input para validación en tiempo real
            nameInput.addEventListener('input', function() {
                const validation = validateField(this.value, 'name');
                updateFeedback('nameFeedback', validation, this);
                updateFormState();
            });
            
            emailInput.addEventListener('input', function() {
                const validation = validateField(this.value, 'email');
                updateFeedback('emailFeedback', validation, this);
                updateFormState();
            });
            
            subjectInput.addEventListener('input', function() {
                const validation = validateField(this.value, 'text', 3);
                updateFeedback('subjectFeedback', validation, this);
                updateFormState();
            });
            
            messageInput.addEventListener('input', function() {
                const validation = validateField(this.value, 'text', 10);
                updateFeedback('messageFeedback', validation, this);
                updateFormState();
            });
            
            // Función auxiliar para actualizar el feedback visual
            function updateFeedback(feedbackElementId, validation, inputElement) {
                const feedbackElement = document.getElementById(feedbackElementId);
                feedbackElement.textContent = validation.message;
                feedbackElement.className = 'feedback-message ' + 
                    (validation.isValid ? 'feedback-success' : 'feedback-error');

                inputElement.classList.remove('valid', 'invalid');

            }
